   851→   851→        <div class="limit-item reveal d1"><span class="limit-no">01</span><div><h3>False positives happen</h3><p>Long, parameter-heavy links from Google, GitHub or password managers can score “suspicious”. Context matters — when in doubt, navigate manually.</p></div></div>
   852→   852→        <div class="limit-item reveal d2"><span class="limit-no">02</span><div><h3>False negatives happen</h3><p>A freshly-registered phishing domain with clean strings can sail through lexical checks. This is a first line of defense, not the last.</p></div></div>

   854→   854→        <div class="limit-item reveal d1"><span class="limit-no">04</span><div><h3>HTTPS ≠ safe</h3><p>Attackers happily buy certificates.    855→   855→        <div class="limit-item reveal d2"><span class="limit-no">05</span><div><h3>Common-password list is finite</h3><p>Our local list covers the most frequently cracked passwords (~150 entries).    856→   856→        <div class="limit-item reveal d3"><span class="limit-no">06</span><div><h3>Meters are estimates</h3><p>Strength meters guide, they don't guarantee.   1069→  1069→  function levenshtein(a, b) {
  1070→  1070→    a = a.toLowerCase(); b = b.toLowerCase();
  1071→  1071→    const m = a.length, n = b.length;
  1072→  1072→    if (Math.abs(m - n) > 2) return Math.max(m, n);
  1073→  1073→    const dp = new Array(n + 1);
  1074→  1074→    for (let j = 0; j <= n; j++) dp[j] = j;
  1075→  1075→    for (let i = 1; i <= m; i++) {
  1076→  1076→      let prev = dp[0]; dp[0] = i;
  1077→  1077→      for (let j = 1; j <= n; j++) {
  1078→  1078→        const tmp = dp[j];
  1079→  1079→        dp[j] = a[i - 1] === b[j - 1] ? prev : 1 + Math.min(prev, dp[j], dp[j - 1]);
  1080→  1080→        prev = tmp;
  1081→  1081→      }
  1082→  1082→    }
  1083→  1083→    return dp[n];
  1084→  1084→  }

  1091→  1091→      .replace(/vv/g, 'w');
  1092→  1092→  }

  1103→  1103→    const add = (level, points, title, detail) => {
  1104→  1104→      findings.push({ level, points, title, detail });
  1105→  1105→      risk += points;
  1106→  1106→    };

  1108→  1108→    let input = (rawInput || '').trim();

  1110→  1110→

  1115→  1115→    }

  1117→  1117→      return {
  1118→  1118→        verdict: 'RISKY', score: 100, scheme: input.split(':')[0].toLowerCase(),
  1119→  1119→        host: '—', labels: 0, port: '—',

  1121→  1121→        tips: ['Delete the message this link came from.', 'Never copy-paste javascript: or data: URLs.']
  1122→  1122→      };
  1123→  1123→    }
  1124→  1124→

  1126→  1126→    try {
  1127→  1127→      url = new URL(input);

  1130→  1130→    }

  1538→  1538→  function evaluatePassword(pw) {

  1545→  1545→    let points = 0;
  1546→  1546→    if (len === 0) points = 0;
  1547→  1547→    else if (len < 8) points = 8 + len * 2;
  1548→  1548→    else if (len < 12) points = 34 + (len - 8) * 4;
  1549→  1549→    else if (len < 15) points = 50 + (len - 12) * 7;
  1550→  1550→    else if (len < 20) points = 71 + (len - 15) * 6;
  1551→  1551→    else points = Math.min(100, 95 + (len - 19) * 1.5);
  1552→  1552→
  1553→  1553→    points += Math.min(10, Math.max(0, (entropy - len * 3.5) * 0.25));
  1554→  1554→
  1555→  1555→    if (common.hit) points -= 45;
  1556→  1556→    if (seqRun >= 5) points -= 14;
  1557→  1557→    else if (seqRun >= 3) points -= 6;
  1558→  1558→    if (repeatRun >= 6) points -= 14;
  1559→  1559→    else if (repeatRun >= 3) points -= 5;
  1560→  1560→    if (/^\d+$/.test(pw)) points -= 10;
  1561→  1561→    if (len > 0 && len < 5) points = Math.min(points, 8);

  1571→  1571→
  1572→  1572→    if (singlePool) { points = Math.min(points, 14); }
  1573→  1573→    else if (totalSeqCoverage >= 0.7) { points = Math.min(points, 22); }
  1574→  1574→    else if (totalSeqCoverage >= 0.45) { points = Math.min(points, 32); }
  1575→  1575→    else if (seqCoverage >= 0.6) { points = Math.min(points, 30); }
  1576→  1576→    else if (seqCoverage >= 0.4) { points = Math.min(points, 45); }
  1577→  1577→    else if (repeatCoverage >= 0.6) { points = Math.min(points, 30); }
  1578→  1578→    else if (repeatCoverage >= 0.4) { points = Math.min(points, 45); }

  1584→  1584→    else if (points < 28) { band = 1; label = 'Very Weak'; }
  1585→  1585→    else if (points < 48) { band = 2; label = 'Weak'; }
  1586→  1586→    else if (points < 68) { band = 3; label = 'Fair'; }
  1587→  1587→    else if (points < 86) { band = 4; label = 'Strong'; }
  1588→  1588→    else { band = 5; label = 'Very Strong'; }
  1589→  1589→
  1590→  1590→    const checks = {
  1591→  1591→      len8: len >= 8,
  1592→  1592→      len15: len >= 15,
  1593→  1593→      common: len > 0 && !common.hit,
  1594→  1594→      pattern: len > 0 && seqRun < 3 && repeatRun < 3,
  1595→  1595→      variety: entropy >= 60 && len >= 12
  1596→  1596→    };
  1597→  1597→
  1598→  1598→    let crack = '—';
  1599→  1599→    if (common.hit) {
  1600→  1600→      crack = 'instantly — in common-password lists';
  1601→  1601→    } else if (len > 0) {

  1603→  1603→      crack = humanizeSeconds(guesses / 1e10);
  1604→  1604→    }
  1605→  1605→
  1606→  1606→    const tips = [];
  1607→  1607→    if (len === 0) {
  1608→  1608→      tips.push('Start typing — analysis runs locally, keystroke by keystroke.');
  1609→  1609→    } else {

  1612→  1612→      else if (len < 15) tips.push('Push to 15+ characters — NIST’s single-factor recommendation. Length is the cheapest security you can add.');
  1613→  1613→      if (seqRun >= 3) tips.push('Break up sequences — “qwerty”, “abc”, “123” are tried in the first second of any attack.');
  1614→  1614→      if (repeatRun >= 3) tips.push('Avoid runs of one repeated character or repeating blocks (abcabc).');

  1616→  1616→      if (tips.length === 0 && band >= 4) {

  1618→  1618→        tips.push('This is exactly what a passphrase should look like — long, odd, and memorable to only you.');
  1619→  1619→      } else if (band === 3) {
  1620→  1620→        tips.push('Solid base. Add a few more words to hit passphrase territory (4+ random words).');
  1621→  1621→      }
  1622→  1622→      if (len >= 8 && len < 15) tips.push('Remember: 8 characters is only comfortable when MFA guards the same account.');
  1623→  1623→      if (tips.length < 2) tips.push('A random four-word passphrase (e.g. “violet-trombone-cafeteria-lantern”) beats “P@ssw0rd!” every time.');
  1624→  1624→    }
  1625→  1625→
  1626→  1626→    return { points, band, label, checks, tips, entropy, crack, len };
  1627→  1627→  }

  1629→  1629→  function humanizeSeconds(s) {
  1630→  1630→    if (s < 1) return 'instantly';
  1631→  1631→    const units = [[31557600, 'years'], [2592000, 'months'], [86400, 'days'], [3600, 'hours'], [60, 'minutes'], [1, 'seconds']];
  1632→  1632→    for (const [sec, name] of units) {
  1633→  1633→      if (s >= sec) {
  1634→  1634→        const v = s / sec;
  1635→  1635→        const nice = v >= 100 ? Math.round(v).toLocaleString() : Math.round(v * 10) / 10;
  1636→  1636→        if (name === 'years' && v > 1e6) {
  1637→  1637→          if (v > 1e12) return 'heat-death of the universe';
  1638→  1638→          if (v > 1e9) return 'billions of years';

  1793→  1793→        const r = l.el.getBoundingClientRect();

  1802→  1802→

  1804→  1804→      current += (target - current) * 0.085; // lerp — buttery but cheap
  1805→  1805→      if (Math.abs(target - current) < 0.4) current = target;
  1806→  1806→
